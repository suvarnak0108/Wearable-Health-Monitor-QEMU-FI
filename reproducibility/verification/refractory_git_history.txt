+def run_cmd(cmd, cwd=ROOT, check=True):
+    proc = subprocess.run(cmd, cwd=str(cwd), capture_output=True, text=True)
+    if check and proc.returncode != 0:
+        raise RuntimeError(
+            f"Command failed ({proc.returncode}): {' '.join(map(str, cmd))}\n"
+            f"STDOUT:\n{proc.stdout}\nSTDERR:\n{proc.stderr}"
+        )
+    return proc

+def sha256_file(path: Path) -> str:
+    h = hashlib.sha256()
+    with path.open("rb") as f:
+        for chunk in iter(lambda: f.read(65536), b""):
+            h.update(chunk)
+    return h.hexdigest()

+def synth_gaussian_ecg(t_vec: np.ndarray, beat_times: np.ndarray) -> np.ndarray:
+    amp = [0.12, -0.15, 1.00, -0.22, 0.28]
+    off = [-0.16, -0.035, 0.0, 0.035, 0.22]
+    wid = [0.018, 0.007, 0.0085, 0.007, 0.032]
+    sig = np.zeros_like(t_vec, dtype=np.float64)
+    for tk in beat_times:
+        for a, d, w in zip(amp, off, wid):
+            sig += a * np.exp(-0.5 * ((t_vec - tk - d) / w) ** 2)
+    return sig

+def hr_trajectory_to_beats(hr_100hz: np.ndarray, fs: float = 100.0):
+    n_samples = len(hr_100hz)
+    t = np.arange(n_samples, dtype=np.float64) / fs
+    beats = []
+    cur_t = 0.4
+    while cur_t < t[-1] - 0.1:
+        idx = int(round(cur_t * fs))
+        if 0 <= idx < n_samples:
+            beats.append(idx)
+        cur_hr = float(hr_100hz[min(n_samples - 1, max(0, idx))])
+        cur_t += 60.0 / np.clip(cur_hr, 30.0, 220.0)
+    beats_arr = np.array(beats, dtype=np.int32)
+    ecg = synth_gaussian_ecg(t, beats_arr / fs)
+    return ecg, beats_arr

+def make_1hz_aux(true_hr_100hz: np.ndarray, seed: int, lag_samples: int = 150, jitter_sd: float = 1.8):
+    rng = np.random.default_rng(seed)
+    n = len(true_hr_100hz)
+    aux_hr = np.zeros(n, dtype=np.float64)
+    spo2 = np.zeros(n, dtype=np.float64)
+    cur_hr = float(np.round(true_hr_100hz[0] + rng.normal(0.0, jitter_sd)))
+    cur_spo2 = 98.0
+    for i in range(n):
+        if i % 100 == 0:
+            lag_i = max(0, i - lag_samples)
+            cur_hr = float(np.round(true_hr_100hz[lag_i] + rng.normal(0.0, jitter_sd)))
+            cur_spo2 = float(int(np.clip(np.round(98.0 + rng.normal(0.0, 0.4)), 96, 99)))
+        aux_hr[i] = cur_hr
+        spo2[i] = cur_spo2
+    return aux_hr, spo2

+def biquad_bandpass_3stage(x: np.ndarray, fs: float = 100.0) -> np.ndarray:
+    sos = butter(2, [5.0, 15.0], btype="bandpass", fs=fs, output="sos").astype(np.float32)
+    y = sosfilt(sos, np.asarray(x, dtype=np.float32))
+    return np.asarray(y, dtype=np.float32)

+def detect_rpeaks_adaptive(y3: np.ndarray, r_min: int = 30, spki_init: float = 0.20, floor_min: float = 0.012):
+    e = (y3.astype(np.float32)) ** 2
+    spki = float(spki_init)
+    npki = 0.0
+    theta = max(floor_min, npki + 0.40 * (spki - npki))
+    p_last = 0
+    peaks = []
+    for n in range(2, len(e)):
+        if e[n] < e[n - 1] and e[n - 1] >= e[n - 2]:
+            c = n - 1
+            e_peak = float(e[c])
+            dn = c - p_last
+            if (p_last == 0 or dn >= r_min) and e_peak > theta:
+                spki = 0.125 * e_peak + 0.875 * spki
+                theta = max(floor_min, npki + 0.40 * (spki - npki))
+                p_last = c
+                peaks.append(c)
+            elif e_peak > 0.0005:
+                npki = 0.125 * e_peak + 0.875 * npki
+                theta = max(floor_min, npki + 0.40 * (spki - npki))
+    return np.array(peaks, dtype=np.int32)

+def rolling_9beat_hr(peaks: np.ndarray, n_len: int, fs: float = 100.0) -> np.ndarray:
+    out = np.zeros(n_len, dtype=np.float64)
+    if len(peaks) < 2:
+        return out
+    hr_inst = 60.0 / np.maximum(np.diff(peaks) / fs, 1e-6)
+    p_idx = 0
+    for n in range(n_len):
+        while p_idx + 1 < len(peaks) and peaks[p_idx + 1] <= n:
+            p_idx += 1
+        if p_idx >= 1:
+            win = hr_inst[max(0, p_idx - 8) : p_idx]
+            if len(win) > 0:
+                out[n] = float(np.median(win))
+    return out

+def classify_faults_5stage(dt_ms, x, hr_s, spo2, hr_ecg_rolling):
+    """Codes: 0=NONE, 1=COMM_BURST, 2=MISSING_SAMPLE, 3=OUT_OF_RANGE, 4=STALE_DATA, 5=HR_INCONSISTENCY."""
+    n_len = len(x)
+    codes = np.zeros(n_len, dtype=np.int32)
+    k_trace = np.zeros(n_len, dtype=np.int32)
+    k_ecg = 0
+    x_prev = float(x[0]) if n_len > 0 else 0.0
+
+    for n in range(n_len):
+        dt = float(dt_ms[n])
+        xn = float(x[n])
+        hrs = float(hr_s[n])
+        sp = float(spo2[n])
+        hecg = float(hr_ecg_rolling[n])
+
+        if dt >= 50.0:
+            k_ecg = 0
+            k_trace[n] = k_ecg
+            codes[n] = 1
+            x_prev = xn
+            continue
+        if dt > 15.0:
+            k_ecg = 0
+            k_trace[n] = k_ecg
+            codes[n] = 2
+            x_prev = xn
+            continue
+        if (hrs < 30.0 or hrs > 200.0) or (sp < 70.0 or sp > 100.0) or np.isnan(hrs) or np.isnan(xn):
+            k_trace[n] = k_ecg
+            codes[n] = 3
+            x_prev = xn
+            continue
+
+        if abs(xn - x_prev) < 1e-6:
+            k_ecg = min(250, k_ecg + 1)
+        else:
+            k_ecg = 0
+        x_prev = xn
+        k_trace[n] = k_ecg
+
+        if k_ecg >= 150:
+            codes[n] = 4
+            continue
+        if hecg > 0.0 and abs(hrs - hecg) > 12.0:
+            codes[n] = 5
+            continue
+        codes[n] = 0
+    return codes, k_trace

+def windowed_10s_hr_rr_filtered(peaks: np.ndarray, n_len: int = 6000, win_size: int = 1000, fs: float = 100.0) -> np.ndarray:
+    n_wins = n_len // win_size
+    hr_w = np.zeros(n_wins, dtype=np.float64)
+    last_hr = 75.0
+    for w in range(n_wins):
+        s0, s1 = w * win_size, (w + 1) * win_size
+        w_peaks = peaks[(peaks >= s0) & (peaks < s1)]
+        if len(w_peaks) >= 2:
+            rr = np.diff(w_peaks).astype(np.float64)
+            valid = rr[(rr >= 0.30 * fs) & (rr <= 1.50 * fs)]
+            if len(valid) > 0:
+                last_hr = (60.0 * fs) / float(np.median(valid))
+        hr_w[w] = last_hr
+    return hr_w

+def aami_beat_metrics(true_peaks: np.ndarray, det_peaks: np.ndarray, tol_samples: int = 15):
+    matched_t = set()
+    tp = 0
+    for d in det_peaks:
+        diffs = np.abs(true_peaks - d)
+        if len(diffs) == 0:
+            continue
+        idx = int(np.argmin(diffs))
+        if diffs[idx] <= tol_samples and idx not in matched_t:
+            matched_t.add(idx)
+            tp += 1
+    fp = int(len(det_peaks) - tp)
+    fn = int(len(true_peaks) - tp)
+    sens = (tp / len(true_peaks) * 100.0) if len(true_peaks) > 0 else 100.0
+    ppv = (tp / len(det_peaks) * 100.0) if len(det_peaks) > 0 else 100.0
+    f1 = (2.0 * sens * ppv / (sens + ppv)) if (sens + ppv) > 0 else 0.0
+    return {"tp": tp, "fp": fp, "fn": fn, "sens": sens, "ppv": ppv, "f1": f1}

+def run_unified_noise_and_refractory():
+    """Item 1, 4, 5: Single consistent run for sigma=0..0.35, 480ms vs 300ms, and sigma=0.08 non-transition analysis."""
+    step_vals = [70.0, 84.0, 75.0, 95.0, 80.0, 70.0]

+    dt_ms = np.full(6000, 10, dtype=np.int32)
+    gt = np.zeros(6000, dtype=bool)
+
+    # Identify step-transition samples (within 2.0s [200 samples] after each 10s boundary)
+    trans_mask = np.zeros(6000, dtype=bool)
+    for boundary in [1000, 2000, 3000, 4000, 5000]:
+        trans_mask[boundary : min(6000, boundary + 200)] = True
+
+    sigmas = [0.0, 0.08, 0.15, 0.25, 0.35]
+    report = {"300ms_Rmin30": {}, "480ms_Rmin48": {}}
+
+    for label, r_min in [("300ms_Rmin30", 30), ("480ms_Rmin48", 48)]:
+        for sigma in sigmas:
+            peaks_l, fpr_l, mae_l = [], [], []
+            nontrans_abs_diff = []
+            nontrans_gt12_frac = []
+            for tr in range(20):
+                seed = 1000 + tr
+                rng = np.random.default_rng(seed)
+                ecg_n = ecg_step + (rng.normal(0.0, sigma, size=6000) if sigma > 0 else 0.0)
+                aux_hr, spo2 = make_1hz_aux(hr_step, seed=seed)
+                y3 = biquad_bandpass_3stage(ecg_n, fs=100.0)
+                peaks = detect_rpeaks_adaptive(y3, r_min=r_min)

+                codes, _ = classify_faults_5stage(dt_ms, ecg_n, aux_hr, spo2, hr_roll)
+                fp = int(np.sum((codes != 0) & (~gt)))
+                tn = int(np.sum((codes == 0) & (~gt)))
+                fpr = fp / (fp + tn) * 100.0
+                h_rr = windowed_10s_hr_rr_filtered(peaks, 6000, 1000, 100.0)
+                mae = float(np.mean(np.abs(h_rr - w_true)))
+
+                peaks_l.append(len(peaks))
+                fpr_l.append(fpr)
+                mae_l.append(mae)
+
+                valid_nt = (~trans_mask) & (hr_roll > 0)
+                if np.any(valid_nt):
+                    diffs = np.abs(aux_hr[valid_nt] - hr_roll[valid_nt])
+                    nontrans_abs_diff.append(float(np.mean(diffs)))
+                    nontrans_gt12_frac.append(float(np.mean(diffs > 12.0) * 100.0))
+
+            report[label][f"sigma_{sigma:.2f}"] = {
+                "peaks_mean": float(np.mean(peaks_l)),
+                "peaks_sd": float(np.std(peaks_l, ddof=1)),
+                "fpr_mean": float(np.mean(fpr_l)),
+                "fpr_sd": float(np.std(fpr_l, ddof=1)),
+                "mae_mean": float(np.mean(mae_l)),
+                "mae_sd": float(np.std(mae_l, ddof=1)),
+                "nontrans_mean_abs_hr_diff_bpm": float(np.mean(nontrans_abs_diff)) if nontrans_abs_diff else None,
+                "nontrans_frac_gt12_pct": float(np.mean(nontrans_gt12_frac)) if nontrans_gt12_frac else None,
+            }
+
+    (OUT_DIR / "unified_noise_and_refractory.json").write_text(json.dumps(report, indent=2))

+        f1s_clean, maes_triple = [], []
+        for idx_r, rec_id in enumerate(records):
+            rec = wfdb.rdrecord(rec_id, pn_dir="mitdb", sampfrom=0, sampto=21600)
+            ann = wfdb.rdann(rec_id, "atr", pn_dir="mitdb", sampfrom=0, sampto=21600)
+            sig = rec.p_signal[:, 0]
+            ecg = resample(sig, 6000)
+            med = np.median(ecg)
+            scale = np.percentile(np.abs(ecg - med), 99)
+            ecg_norm = (ecg - med) / (scale if scale > 1e-9 else 1.0)
+
+            true_peaks = np.array([int(round(s * 6000.0 / 21600.0)) for s, sym in zip(ann.sample, ann.symbol) if sym in beat_syms], dtype=np.int32)
+            det_clean = detect_rpeaks_adaptive(biquad_bandpass_3stage(ecg_norm, fs=100.0), r_min=r_min)
+            bm = aami_beat_metrics(true_peaks, det_clean, tol_samples=15)
+
+            # Triple compound clinical run
+            ecg_tr = ecg_norm + 0.15 * nstdb_em
+            aux_hr, spo2 = make_1hz_aux(np.full(6000, 75.0), seed=900 + idx_r)
+            dt_tr = np.full(6000, 10, dtype=np.int32)
+            for b in range(10):
+                s = 300 + b * 500
+                dt_tr[s : s + 10] = 60
+            aux_hr[2000:3500] += 22.0
+
+            det_tr = detect_rpeaks_adaptive(biquad_bandpass_3stage(ecg_tr, fs=100.0), r_min=r_min)
+            h_tr = windowed_10s_hr_rr_filtered(det_tr, 6000, 1000, 100.0)
+            h_ref = windowed_10s_hr_rr_filtered(true_peaks, 6000, 1000, 100.0)
+            mae_tr = float(np.mean(np.abs(h_tr - h_ref)))
+
+            f1s_clean.append(bm["f1"])
+            maes_triple.append(mae_tr)
+            per_rec.append({"record": rec_id, "clean_beat_f1": bm["f1"], "triple_rr_mae": mae_tr})
+
+        out[label] = {
+            "clean_beat_f1_mean": float(np.mean(f1s_clean)),
+            "clean_beat_f1_sd": float(np.std(f1s_clean, ddof=1)),
+            "triple_rr_mae_mean": float(np.mean(maes_triple)),
+            "triple_rr_mae_sd": float(np.std(maes_triple, ddof=1)),
+            "per_record": per_rec,
+        }
+
+    (OUT_DIR / "refractory_480ms_clinical_beats.json").write_text(json.dumps(out, indent=2))
+    return out
