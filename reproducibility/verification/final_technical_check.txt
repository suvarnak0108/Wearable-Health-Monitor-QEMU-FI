=== Fault detector ===
#include "wearable_monitor.h"

static inline float fabsf_local(float x) { return (x < 0.0f) ? -x : x; }

void Fault_Init(FaultDetectorState *st) {
    st->prev_ecg = 0.0f;
    st->prev_hr = 0.0f;
    st->prev_spo2 = 0.0f;
    st->stale_ecg = 0U;
    st->stale_hr = 0U;
    st->stale_spo2 = 0U;
}

FaultCode Fault_Detect_Full(FaultDetectorState *st, uint32_t dt_ms, float ecg,
                            float sensor_hr, float spo2, float ecg_hr) {
    if (dt_ms >= 50U) {
        st->stale_ecg = 0U;
        return FAULT_COMM_BURST;
    }
    if (dt_ms > 15U) {
        st->stale_ecg = 0U;
        return FAULT_MISSING_SAMPLE;
    }
    if (sensor_hr != sensor_hr || spo2 != spo2 || ecg != ecg ||
        sensor_hr < 30.0f || sensor_hr > 200.0f ||
        spo2 < 70.0f || spo2 > 100.0f) {
        return FAULT_OUT_OF_RANGE;
    }
    /* High-rate (100 Hz) ECG ADC freeze >= 150 samples (1.5 s > 1 full cardiac cycle) */
    if (fabsf_local(ecg - st->prev_ecg) < 1e-6f) {
        if (st->stale_ecg < 250U) st->stale_ecg++;
    } else {
        st->stale_ecg = 0U;
    }
    st->prev_ecg = ecg;
    if (st->stale_ecg >= 150U) {
        return FAULT_STALE_DATA;
    }
    /* Cross-modal HR consistency check */
    if (ecg_hr > 0.0f && fabsf_local(sensor_hr - ecg_hr) > 12.0f) {
        return FAULT_HR_INCONSISTENCY;
    }
    return FAULT_NONE;
}
=== R-peak detector ===
#include "wearable_monitor.h"

/* 300 ms refractory period @ 100 Hz -> supports up to 200 BPM */
#define REFRACTORY_SAMPLES 30U
#define MIN_ENERGY_FLOOR   0.012f

void RPeak_Init(RPeakDetectorState *st) {
    st->spki = 0.20f;
    st->npki = 0.01f;
    st->threshold = 0.086f;
    st->last_peak_idx = 0U;
    st->prev_e1 = 0.0f;
    st->prev_e2 = 0.0f;
}

uint8_t RPeak_Process(RPeakDetectorState *st, float y_filt, uint32_t idx) {
    /* Full-wave quadratic energy captures both upright and inverted PVCs */
    float e = y_filt * y_filt;
    uint8_t detected = 0U;

    if (idx >= 2U && e < st->prev_e1 && st->prev_e1 >= st->prev_e2) {
        uint32_t cand = idx - 1U;
        float e_peak = st->prev_e1;
        uint32_t dt = cand - st->last_peak_idx;

        if ((st->last_peak_idx == 0U || dt >= REFRACTORY_SAMPLES) && e_peak > st->threshold) {
            st->spki = 0.125f * e_peak + 0.875f * st->spki;
            float th = st->npki + 0.40f * (st->spki - st->npki);
            st->threshold = (th < MIN_ENERGY_FLOOR) ? MIN_ENERGY_FLOOR : th;
            st->last_peak_idx = cand;
            detected = 1U;
        } else if (e_peak > 0.0005f) {
            st->npki = 0.125f * e_peak + 0.875f * st->npki;
            float th = st->npki + 0.40f * (st->spki - st->npki);
            st->threshold = (th < MIN_ENERGY_FLOOR) ? MIN_ENERGY_FLOOR : th;
        }
    }

    st->prev_e2 = st->prev_e1;
    st->prev_e1 = e;
    return detected;
}
